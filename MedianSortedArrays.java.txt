class Solution {
    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        int n = nums1.length;
        int m = nums2.length;
        int[] arr = new int[n+m];
        int a1 = 0,a2 = 0,a3 = 0;
        while(a1<n && a2<m){
            if(nums1[a1]<nums2[a2]){
                arr[a3] = nums1[a1];
                a1++;
                a3++;
            }else{
                arr[a3] = nums2[a2];
                a2++;
                a3++;
            }
        }
        while(a1<n){
            arr[a3] = nums1[a1];
            a1++;
            a3++;
        }
        while(a2<m){
            arr[a3] = nums2[a2];
            a2++;
            a3++;
        }
        int len = arr.length;
        if(len%2==0){
            return (arr[len/2 - 1]+arr[len/2])/2.0;
        }else{
            return  arr[len/2];
        }
    }
}
