import java.util.*;
class Solution {
    public int[] twoSum(int[] nums, int target) {
        int[] arr = new int[2];
        int n = nums.length;
        int idx = 0;
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                if(nums[i]+nums[j]==target){
                    arr[idx++] = i;
                    arr[idx++] = j;
                    break;
                }
            }
        }
        return arr;
    }
}
